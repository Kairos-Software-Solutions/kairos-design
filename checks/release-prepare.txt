set -euo pipefail

package_version="$(node -p "require('./package.json').version")"

if [[ ! "$package_version" =~ ^[0-9]+\.[0-9]+\.[0-9]+(-[0-9A-Za-z.-]+)?$ ]]; then
  echo "package.json contains an invalid release version: $package_version"
  exit 1
fi

package_tag="v$package_version"

if [[ "$GITHUB_REF_TYPE" == "tag" ]]; then
  if [[ "$GITHUB_REF_NAME" != "$package_tag" ]]; then
    echo "Tag $GITHUB_REF_NAME does not match package.json version $package_version."
    exit 1
  fi

  echo "publish=true" >> "$GITHUB_OUTPUT"
  echo "create_tag=false" >> "$GITHUB_OUTPUT"
  exit 0
fi

if git show-ref --verify --quiet "refs/tags/$package_tag"; then
  echo "publish=false" >> "$GITHUB_OUTPUT"
  echo "create_tag=false" >> "$GITHUB_OUTPUT"
  echo "$package_tag already exists. Nothing to release." >> "$GITHUB_STEP_SUMMARY"
else
  echo "publish=true" >> "$GITHUB_OUTPUT"
  echo "create_tag=true" >> "$GITHUB_OUTPUT"
  echo "tag=$package_tag" >> "$GITHUB_OUTPUT"
fi
--- end script ---
